cmake_minimum_required(VERSION 3.20)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED True)
project(syncstoolkit)
include_directories(${CMAKE_SOURCE_DIR})
link_directories(${CMAKE_SOURCE_DIR})
add_definitions(-DLOGLEVEL=4 -DNOMINMAX)
file(GLOB src CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/*.h" "${CMAKE_SOURCE_DIR}/*.cpp")
add_library(syncstoolkit SHARED ${src})
target_link_libraries(syncstoolkit libMinHook.x64 modsettings setupapi)
